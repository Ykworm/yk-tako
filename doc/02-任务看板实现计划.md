> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新建 `internal/taskboard`（TaskCreate/TaskUpdate/TaskList/TaskGet 记账工具 + middleware 正本 + session 镜像），收敛出唯一入口 `cmd/yktako`，删除其余五个入口与 `internal/planexec`，重写文档。

**Architecture:** 三层分离（doc/01）：Plan 是提示词约定的口述文档；Tasks 是 taskboard 看板，正本存 middleware 自持 Store（跨 REPL 轮次存活），每次变更后 `adk.AddSessionValue` 全量镜像（interrupt 时随 runContext gob 进 checkpoint）；Actions 是现有工具，无外键。宿主是裸 ChatModelAgent，Handlers 链 = [taskboard] + guard 三件套。

**Tech Stack:** Go 1.25 + eino v0.9.21（adk / compose / schema / components/tool/utils）。

```go
func TestStoreConcurrentCreate(t *testing.T) {
	s := NewStore()
	var wg sync.WaitGroup
	ids := make(chan string, 100)
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			ids <- s.Create(fmt.Sprintf("任务%d", i), "", "").ID
		}(i)
	}
	wg.Wait()
	close(ids)
	seen := map[string]bool{}
	for id := range ids {
		if seen[id] {
			t.Fatalf("id 重复: %s", id)
		}
		seen[id] = true
	}
	if len(seen) != 100 {
		t.Fatalf("应有 100 个唯一 id，实际 %d", len(seen))
	}
	// 按数值排序比上下限：字符串排序会把 "99" 排在 "100" 之后，比不出上限。
	nums := make([]int, 0, len(seen))
	for id := range seen {
		n, err := strconv.Atoi(id)
		if err != nil {
			t.Fatalf("id 不是数字: %q", id)
		}
		nums = append(nums, n)
	}
	sort.Ints(nums)
	if nums[0] != 1 || nums[len(nums)-1] != 100 {
		t.Fatalf("id 应覆盖 1..100: %d ... %d", nums[0], nums[len(nums)-1])
	}
}
```

```go
// Package taskboard 是任务看板：TaskCreate/TaskUpdate/TaskList/TaskGet 四个记账工具、
// middleware 注入与看板投影。本文件是数据模型与存储正本（Store）；
// 工具与输出格式见 tools.go，middleware 与提示词约定见 middleware.go。
//
// 三层分离（doc/01）：Plan 是给人审的文档，Tasks 是带 taskId 的施工看板，
// Actions 是干活工具。本包只管 Tasks 层。
package taskboard

import (
	"encoding/gob"
	"fmt"
	"sync"
)

// init 把快照类型注册给 gob：session.Values 是 map[string]any，interrupt 时 eino 把整个
// runContext（含 session）gob 进 checkpoint，接口字段背后的具体类型必须注册
// （eino 自己注册 []byte 同款，adk/interrupt.go init）。
func init() {
	gob.Register([]Task{})
}

// SessionKey 是看板镜像在 session 里的键。每次变更后全量快照写入；
// interrupt 时随 runContext 落盘（adk/runctx.go 的 runSession 有
// "CheckpointSchema: persisted via serialization.RunCtx (gob)" 标注）。
const SessionKey = "taskboard_session_key_tasks"

// Status 是任务状态：pending → in_progress → completed，不再需要时用 deleted。
type Status string

// 四个合法状态值，与 eino plantask / Claude Tasks 对齐。
const (
	StatusPending    Status = "pending"
	StatusInProgress Status = "in_progress"
	StatusCompleted  Status = "completed"
	StatusDeleted    Status = "deleted"
)

// validStatus 校验 TaskUpdate 传入的 status 字符串。
func validStatus(s string) bool {
	switch Status(s) {
	case StatusPending, StatusInProgress, StatusCompleted, StatusDeleted:
		return true
	}
	return false
}

// Task 是一条看板行。taskId 只寻址看板行：与 plan 条目、tool_call 都无外键。
type Task struct {
	ID          string `json:"id"`
	Subject     string `json:"subject"`
	Description string `json:"description,omitempty"`
	ActiveForm  string `json:"activeForm,omitempty"`
	Status      Status `json:"status"`
}

// Patch 是 TaskUpdate 的可改字段。空串表示「不动」（所以字段无法清成空串，是有意取舍）。
type Patch struct {
	Subject     string
	Description string
	ActiveForm  string
	Status      string
}

// Store 是看板正本：middleware 自持，agent 建一次，跨 REPL 轮次存活。
// 所有方法加锁，同轮并行的工具调用也写不坏。
type Store struct {
	mu     sync.Mutex
	tasks  map[string]*Task
	order  []string // 创建顺序，看板按它排
	nextID int
}

// NewStore 建一个空看板，id 从 "1" 开始自增。
func NewStore() *Store {
	return &Store{tasks: map[string]*Task{}, nextID: 1}
}

// Create 建一条 pending 任务，返回副本（调用方改不到内部）。
func (s *Store) Create(subject, description, activeForm string) Task {
	s.mu.Lock()
	defer s.mu.Unlock()
	id := fmt.Sprintf("%d", s.nextID)
	s.nextID++
	t := &Task{ID: id, Subject: subject, Description: description, ActiveForm: activeForm, Status: StatusPending}
	s.tasks[id] = t
	s.order = append(s.order, id)
	return *t
}

// Update 按 id patch；id 不存在或 status 非法时报错。
func (s *Store) Update(id string, p Patch) (Task, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t, ok := s.tasks[id]
	if !ok {
		return Task{}, fmt.Errorf("任务 #%s 不存在", id)
	}
	if p.Status != "" {
		if !validStatus(p.Status) {
			return Task{}, fmt.Errorf("非法 status %q：只能是 pending/in_progress/completed/deleted", p.Status)
		}
		t.Status = Status(p.Status)
	}
	if p.Subject != "" {
		t.Subject = p.Subject
	}
	if p.Description != "" {
		t.Description = p.Description
	}
	if p.ActiveForm != "" {
		t.ActiveForm = p.ActiveForm
	}
	return *t, nil
}

// Get 读单条；不存在报错。
func (s *Store) Get(id string) (Task, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t, ok := s.tasks[id]
	if !ok {
		return Task{}, fmt.Errorf("任务 #%s 不存在", id)
	}
	return *t, nil
}

// List 返回未 deleted 的任务，按创建序；返回副本切片。
func (s *Store) List() []Task {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make([]Task, 0, len(s.order))
	for _, id := range s.order {
		if t := s.tasks[id]; t.Status != StatusDeleted {
			out = append(out, *t)
		}
	}
	return out
}

// Snapshot 返回全量任务（含 deleted），给 session 镜像用。
func (s *Store) Snapshot() []Task {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make([]Task, 0, len(s.order))
	for _, id := range s.order {
		out = append(out, *s.tasks[id])
	}
	return out
}
```

```go
	updateDesc = `按 taskId 更新任务（只传要改的字段）。纪律：动手前先把对应任务标 in_progress（同时只保持一条）；做完一项立刻标 completed，不要攒到最后一起勾——测试未通过、只做了一半、有未解决报错时不准标 completed；不要的任务标 deleted。卡住时改用 TaskCreate 建一条 blocker 任务说明卡点。`
	listDesc   = `列出看板上全部未删除任务（含状态），按创建顺序。`
	getDesc    = `按 taskId 读一条任务的完整信息（标题、描述、状态）。`
)

// makeTools 生成四个记账工具，操作同一份 store 正本。
// 每次变更后把全量快照写进 session（adk.AddSessionValue）：
// interrupt 时镜像随 runContext gob 进 checkpoint（机制见 taskboard.go 的 SessionKey 注释）；
// 无 session 的 ctx（单测直调）下镜像是静默空操作。
func makeTools(store *Store) ([]tool.BaseTool, error) {
	// mirror 把当前全量快照写进 session。
	mirror := func(ctx context.Context) {
		adk.AddSessionValue(ctx, SessionKey, store.Snapshot())
	}
	// withBoard 拼标准输出：回执 + 看板 + 催更块。
	withBoard := func(receipt string) string {
		list := store.List()
		return receipt + "\n\n" + board(list) + nudge(list)
	}

	create, err := utils.InferTool(ToolCreate, createDesc,
		func(ctx context.Context, args createArgs) (string, error) {
			t := store.Create(args.Subject, args.Description, args.ActiveForm)
			mirror(ctx)
			return withBoard(fmt.Sprintf("已创建任务 #%s「%s」", t.ID, t.Subject)), nil
		})
	if err != nil {
		return nil, err
	}
	update, err := utils.InferTool(ToolUpdate, updateDesc,
		func(ctx context.Context, args updateArgs) (string, error) {
			t, err := store.Update(args.TaskID, Patch{
				Subject: args.Subject, Description: args.Description,
				ActiveForm: args.ActiveForm, Status: args.Status,
			})
			if err != nil {
				return "", err
			}
			mirror(ctx)
			return withBoard(fmt.Sprintf("已更新任务 #%s（当前状态：%s）", t.ID, t.Status)), nil
		})
	if err != nil {
		return nil, err
	}
	list, err := utils.InferTool(ToolList, listDesc,
		func(ctx context.Context, _ listArgs) (string, error) {
			l := store.List()
			return board(l) + nudge(l), nil
		})
	if err != nil {
		return nil, err
	}
	get, err := utils.InferTool(ToolGet, getDesc,
		func(ctx context.Context, args getArgs) (string, error) {
			t, err := store.Get(args.TaskID)
			if err != nil {
				return "", err
			}
			receipt := fmt.Sprintf("#%s「%s」状态 %s", t.ID, t.Subject, t.Status)
			if t.Description != "" {
				receipt += "\n描述：" + t.Description
			}
			return withBoard(receipt), nil
		})
	if err != nil {
		return nil, err
	}
	return []tool.BaseTool{create, update, list, get}, nil
}

// statusIcon 给看板行的状态图标。
func statusIcon(s Status) string {
	switch s {
	case StatusCompleted:
		return "✅"
	case StatusInProgress:
		return "🔄"
	default:
		return "⏳"
	}
}

// board 渲染紧凑看板；空板返回占位行。
func board(tasks []Task) string {
	if len(tasks) == 0 {
		return "📋 任务看板（空）"
	}
	done := 0
	for _, t := range tasks {
		if t.Status == StatusCompleted {
			done++
		}
	}
	var b strings.Builder
	fmt.Fprintf(&b, "📋 任务看板（%d/%d 已完成）", done, len(tasks))
	for _, t := range tasks {
		subject := t.Subject
		if t.Status == StatusInProgress && t.ActiveForm != "" {
			subject = t.ActiveForm
		}
		fmt.Fprintf(&b, "\n  %s #%s %s", statusIcon(t.Status), t.ID, subject)
	}
	return b.String()
}

// nudge 是催更块：还有未完成项时追加在工具输出尾部；全部完成返回空串。
func nudge(tasks []Task) string {
	remaining := 0
	for _, t := range tasks {
		if t.Status == StatusPending || t.Status == StatusInProgress {
			remaining++
		}
	}
	if remaining == 0 {
		return ""
	}
	return fmt.Sprintf(`
⚠️ 还有 %d 项未完成。要求：
- 动手前先把要做的任务标成 in_progress（同时只保持一条）；
- 做完一项立刻标 completed，不要攒到最后一起勾；
- 全部完成前不要输出最终总结。`, remaining)
}
```

```go
// drain 收完事件流，有错误事件立即失败。
func drain(t *testing.T, iter *adk.AsyncIterator[*adk.AgentEvent]) []*adk.AgentEvent {
	t.Helper()
	var events []*adk.AgentEvent
	for {
		ev, ok := iter.Next()
		if !ok {
			return events
		}
		if ev.Err != nil {
			t.Fatalf("事件流出错: %v", ev.Err)
		}
		events = append(events, ev)
	}
}
```

```go
// yk-tako 唯一入口：裸 ReAct + 任务看板（taskboard）+ L1 三件套守卫。
//
// 组装（doc/01 §5）：ChatModelAgent 的 Handlers 链 = [taskboard] + guard 三件套；
// 看板正本在 taskboard middleware 里跨轮存活，镜像随 interrupt 落 checkpoint。
package main

import (
	"context"
	"fmt"
	"os"
	"strings"

	"github.com/cloudwego/eino/adk"
	"github.com/cloudwego/eino/compose"
	"github.com/cloudwego/eino/schema"
	"yk-tako-eino/internal/adkrun"
	"yk-tako-eino/internal/config"
	"yk-tako-eino/internal/guard"
	"yk-tako-eino/internal/repl"
	"yk-tako-eino/internal/taskboard"
	"yk-tako-eino/internal/tools"
)

// main 建模型、工具、看板与守卫，进 REPL 循环；审批走 interrupt 加 Resume。
func main() {
	ctx := context.Background()
	chatModel, ref := config.MustOpen(ctx)
	cwd, err := os.Getwd()
	if err != nil {
		repl.Exitf("%v", err)
	}
	bash, err := tools.Bash(ctx)
	if err != nil {
		repl.Exitf("bash: %v", err)
	}
	files, err := tools.Files(ctx)
	if err != nil {
		repl.Exitf("files: %v", err)
	}
	all := append(tools.MustBasic(ctx), bash)
	all = append(all, files...)

	// 看板：正本在 store（跨轮存活），middleware 注入四个记账工具。
	boardMW, err := taskboard.New(taskboard.NewStore())
	if err != nil {
		repl.Exitf("taskboard: %v", err)
	}

	engine := guard.NewEngine(guard.ModeApprove, nil)
	handlers := append([]adk.ChatModelAgentMiddleware{boardMW}, guard.Handlers(engine, guard.Options{WorkspaceRoot: cwd})...)
	agent, err := adk.NewChatModelAgent(ctx, &adk.ChatModelAgentConfig{
		Name:        config.AgentName(),
		Description: "带任务看板与工具安全层的工作伙伴",
		Instruction: config.SystemPrompt() + "\n\n" + taskboard.Instruction,
		Model:       chatModel,
		ToolsConfig: adk.ToolsConfig{
			// ExecuteSequentially：同批工具调用串行，「同批里先 Update 再干活」
			// 降低打架（秩序偏好，不是正确性保证，见 doc/01 §4）。
			ToolsNodeConfig: compose.ToolsNodeConfig{Tools: all, ExecuteSequentially: true},
		},
		Handlers:      handlers,
		MaxIterations: 20,
	})
	if err != nil {
		repl.Exitf("创建 agent: %v", err)
	}
	runner := adk.NewRunner(ctx, adk.RunnerConfig{
		Agent:           agent,
		EnableStreaming: false,
		CheckPointStore: adkrun.NewStore(),
	})

	console := repl.New()
	var history []*schema.Message
	turn := 0
	fmt.Printf("yk-tako。模型 %s。\n", ref)
	fmt.Printf("ToolGuard 模式: %s（/mode 可切换）。输入 quit 退出。\n\n", engine.Mode())
	for {
		input, ok := console.ReadLine("你: ")
		if !ok || input == "quit" || input == "exit" {
			break
		}
		if input == "" {
			continue
		}
		if mode, ok := strings.CutPrefix(input, "/mode "); ok {
			switch guard.Mode(strings.TrimSpace(mode)) {
			case guard.ModeBlock, guard.ModeWarn, guard.ModeApprove:
				engine.Configure(guard.Mode(strings.TrimSpace(mode)))
				fmt.Printf("ToolGuard 模式已切换为: %s\n\n", engine.Mode())
			default:
				fmt.Println("用法: /mode block|warn|require_approval")
			}
			continue
		}
		turn++
		messages := append(append([]*schema.Message{}, history...), schema.UserMessage(input))
		produced, err := adkrun.Run(ctx, runner, agent, fmt.Sprintf("turn-%d", turn), messages, func(info guard.ApprovalInfo) (string, error) {
			fmt.Println("\n⏸️  ═══════════════ 人工审批（interrupt）═══════════════")
			fmt.Printf("工具: %s   参数: %s\n", info.Tool, info.Args)
			if info.Message != "" {
				fmt.Println(info.Message)
			}
			answer, ok := console.ReadLine("批准执行？[y/N] ")
			if !ok {
				return guard.ResumeReject, nil
			}
			answer = strings.ToLower(answer)
			if answer == "y" || answer == "yes" {
				fmt.Println("✅ 已批准，继续执行")
				fmt.Println()
				return guard.ResumeApprove, nil
			}
			fmt.Println("🚫 已拒绝")
			fmt.Println()
			return guard.ResumeReject, nil
		})
		if err != nil {
			fmt.Println("出错了:", err)
			continue
		}
		history = append(history, schema.UserMessage(input))
		history = append(history, produced...)
	}
}
```
